cmake_minimum_required(VERSION 3.20)
project(scollatz VERSION 1.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(SCOLLATZ_ENABLE_OPENCL "Build OpenCL backends when available" ON)
option(SCOLLATZ_BUILD_TESTS "Build tests" ON)
find_package(Threads REQUIRED)
if(SCOLLATZ_ENABLE_OPENCL)
  find_package(OpenCL QUIET)
endif()
add_executable(scollatz src/main.cpp)
add_executable(scollatz-ga src/genetic.cpp)
foreach(target scollatz scollatz-ga)
  target_link_libraries(${target} PRIVATE Threads::Threads)
  target_compile_options(${target} PRIVATE $<$<CXX_COMPILER_ID:GNU,Clang>:-Wall;-Wextra;-Wpedantic;-Wconversion;-Wshadow>)
endforeach()
target_compile_definitions(scollatz PRIVATE SCOLLATZ_KERNEL_PATH="${CMAKE_CURRENT_SOURCE_DIR}/kernels/collatz.cl")
target_compile_definitions(scollatz-ga PRIVATE SCOLLATZ_GA_KERNEL_PATH="${CMAKE_CURRENT_SOURCE_DIR}/kernels/collatz.cl")
if(OpenCL_FOUND)
  foreach(target scollatz scollatz-ga)
    target_link_libraries(${target} PRIVATE OpenCL::OpenCL)
    target_compile_definitions(${target} PRIVATE SCOLLATZ_HAS_OPENCL=1)
  endforeach()
  message(STATUS "OpenCL backends enabled")
else()
  message(STATUS "OpenCL not found; CPU-only build")
endif()
if(SCOLLATZ_BUILD_TESTS)
  enable_testing()
  add_test(NAME cpu_smoke COMMAND scollatz --start 1 --end 10000 --backend cpu --threads 2)
  add_test(NAME argument_validation COMMAND scollatz --start 0 --end 10)
  set_tests_properties(argument_validation PROPERTIES WILL_FAIL TRUE)
  add_test(NAME ga_cpu_smoke COMMAND scollatz-ga --backend cpu --population 64 --generations 3 --islands 4 --min 1 --max 100000 --threads 2 --seed 42 --checkpoint ${CMAKE_CURRENT_BINARY_DIR}/ga.chk --csv ${CMAKE_CURRENT_BINARY_DIR}/ga.csv --json ${CMAKE_CURRENT_BINARY_DIR}/ga.json)
  if(OpenCL_FOUND)
    add_test(NAME ga_opencl_smoke COMMAND scollatz-ga --backend opencl --population 32 --generations 2 --islands 2 --min 1 --max 10000 --seed 42 --checkpoint ${CMAKE_CURRENT_BINARY_DIR}/ga-cl.chk --csv ${CMAKE_CURRENT_BINARY_DIR}/ga-cl.csv --json ${CMAKE_CURRENT_BINARY_DIR}/ga-cl.json)
  endif()
endif()
install(TARGETS scollatz scollatz-ga RUNTIME DESTINATION bin)
